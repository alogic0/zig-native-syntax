module main
import time

pub struct Item {
    value string
    enabled bool
    count int
}

pub fn (item Item) total(delta int) int {
    return item.count + delta
}

fn make_item(name string) Item {
    item := Item{
        value: name
        enabled: true
        count: 42
    }
    return item
}
