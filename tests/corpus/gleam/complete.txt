//// A compact structural Gleam sample.
import gleam/list
import gleam/result
import gleam/string as text

@external(erlang, "erlang", "display")
fn display(value: String) -> Nil

pub type Person {
  Person(name: String, enabled: Bool)
}

const prefix = "Hello\\n"
const answer = 42

pub fn greet(person: Person, loud: Bool) -> String {
  let names = list.map([person.name], fn(name) { name })
  case loud {
    True -> prefix <> names
    False -> "hello"
  }
}

pub fn describe(person: Person) -> String {
  let Person(name, enabled) = person
  use suffix <- result.try(Ok("!"))
  let updated = Person(..person, enabled: False)
  let payload = <<1, answer:size(8), name:utf8>>

  updated.name
  |> text.append(suffix)
  |> text.append(case updated {
    Person(current_name, True) if current_name != "" -> current_name
    Person(_, False) -> "disabled"
  })
}
