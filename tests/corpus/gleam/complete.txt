//// A compact structural Gleam sample.
import gleam/list

@external(erlang, "erlang", "display")
fn display(value: String) -> Nil

pub type Person {
  Person(name: String, enabled: Bool)
}

const prefix = "Hello\\n"
const answer = 42

pub fn greet(person: Person, loud: Bool) -> String {
  let names = list.map([person.name], fn(name) { name })
  case loud {
    True -> prefix <> names
    False -> "hello"
  }
}
