module Geometry # public shapes

abstract type Shape end

struct Circle <: Shape
    radius::Float64
end

function area(circle::Circle; scale::Float64 = 1.0)
    @assert circle.radius >= 0
    message = "area\\n<&>"
    return :ok, π * circle.radius^2 * scale
end

area(radius::Float64) = π * radius^2

norm(δ::Float64, σ²::Float64) = sqrt(δ^2 + σ²)

macro checked(expr)
    quote
        $(esc(expr))
    end
end

enabled = true
end
